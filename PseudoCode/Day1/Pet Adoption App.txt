CLASS Pet
    ATTRIBUTES:
        name
        age
        breed
    
    CONSTRUCTOR Pet(name, age, breed):
        this.name = name
        this.age = age
        this.breed = breed
    
    METHOD displayInfo():
        PRINT "Name: " + name
        PRINT "Age: " + age
        PRINT "Breed: " + breed
END CLASS


CLASS Dog EXTENDS Pet
    ATTRIBUTES:
        isTrained
    
    CONSTRUCTOR Dog(name, age, breed, isTrained):
        CALL Pet(name, age, breed)
        this.isTrained = isTrained
    
    METHOD displayInfo():
        CALL super.displayInfo()
        PRINT "Trained: " + isTrained
END CLASS


CLASS Cat EXTENDS Pet
    ATTRIBUTES:
        isIndoor
    
    CONSTRUCTOR Cat(name, age, breed, isIndoor):
        CALL Pet(name, age, breed)
        this.isIndoor = isIndoor
    
    METHOD displayInfo():
        CALL super.displayInfo()
        PRINT "Indoor Cat: " + isIndoor
END CLASS


CLASS Bird EXTENDS Pet
    ATTRIBUTES:
        canTalk
    
    CONSTRUCTOR Bird(name, age, breed, canTalk):
        CALL Pet(name, age, breed)
        this.canTalk = canTalk
    
    METHOD displayInfo():
        CALL super.displayInfo()
        PRINT "Can Talk: " + canTalk
END CLASS


FUNCTION Main():
    dog1 = NEW Dog("Bruno", 3, "Labrador", true)
    cat1 = NEW Cat("Kitty", 2, "Persian", false)
    bird1 = NEW Bird("Rio", 1, "Parrot", true)

    PRINT "Adopted Pets Info:"
    CALL dog1.displayInfo()
    CALL cat1.displayInfo()
    CALL bird1.displayInfo()
END FUNCTION
