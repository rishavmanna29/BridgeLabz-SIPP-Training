CLASS Flight
    ATTRIBUTES:
        flightNumber
        price
        duration   // in hours
    
    CONSTRUCTOR Flight(flightNumber, price, duration):
        this.flightNumber = flightNumber
        this.price = price
        this.duration = duration
    
    METHOD displayInfo():
        PRINT "Flight: " + flightNumber + ", Price: " + price + ", Duration: " + duration
END CLASS


// COMPARATOR INTERFACE
INTERFACE Comparator
    METHOD compare(f1, f2)
END INTERFACE


// PRICE COMPARATOR
CLASS PriceComparator IMPLEMENTS Comparator
    METHOD compare(f1, f2):
        RETURN f1.price - f2.price
END CLASS


// DURATION COMPARATOR
CLASS DurationComparator IMPLEMENTS Comparator
    METHOD compare(f1, f2):
        RETURN f1.duration - f2.duration
END CLASS


// MERGE SORT IMPLEMENTATION
FUNCTION mergeSort(flights[], comparator):
    IF length of flights <= 1 THEN
        RETURN flights
    
    mid = length(flights) / 2
    left = mergeSort(flights[0..mid-1], comparator)
    right = mergeSort(flights[mid..end], comparator)
    
    RETURN merge(left, right, comparator)


FUNCTION merge(left[], right[], comparator):
    result = []
    i = 0, j = 0
    
    WHILE i < length(left) AND j < length(right):
        IF comparator.compare(left[i], right[j]) <= 0 THEN
            ADD left[i] TO result
            i = i + 1
        ELSE
            ADD right[j] TO result
            j = j + 1
    
    WHILE i < length(left):
        ADD left[i] TO result
        i = i + 1
    
    WHILE j < length(right):
        ADD right[j] TO result
        j = j + 1
    
    RETURN result


// MAIN PROGRAM
FUNCTION Main():
    flights = []
    ADD NEW Flight("AI101", 5000, 3) TO flights
    ADD NEW Flight("6E202", 3500, 2) TO flights
    ADD NEW Flight("UK303", 4500, 4) TO flights
    ADD NEW Flight("SG404", 3000, 5) TO flights
    
    PRINT "--- Sorting by Price ---"
    sortedByPrice = mergeSort(flights, NEW PriceComparator())
    FOR each f IN sortedByPrice:
        f.displayInfo()
    
    PRINT "--- Sorting by Duration ---"
    sortedByDuration = mergeSort(flights, NEW DurationComparator())
    FOR each f IN sortedByDuration:
        f.displayInfo()
END FUNCTION
