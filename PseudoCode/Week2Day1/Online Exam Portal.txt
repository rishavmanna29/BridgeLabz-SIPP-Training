CLASS User
    ATTRIBUTES:
        username : String
        password : String
    
    CONSTRUCTOR User(username, password):
        this.username = username
        this.password = password
    
    METHOD login():
        PRINT "Generic login validation for user."
END CLASS


// ---------------- STUDENT ----------------
CLASS Student EXTENDS User
    ATTRIBUTES:
        enrolled : Boolean
    
    CONSTRUCTOR Student(username, password, enrolled):
        SUPER(username, password)
        this.enrolled = enrolled
    
    METHOD login():   // Overriding
        IF validateCredentials(username, password) AND enrolled == TRUE:
            PRINT "Student " + username + " logged in successfully. Access exams."
        ELSE:
            PRINT "Login failed. Student not registered or not enrolled."
    
    METHOD validateCredentials(username, password):
        // Simulated check (real-world: DB check)
        RETURN password == "student123"
END CLASS


// ---------------- ADMIN ----------------
CLASS Admin EXTENDS User
    ATTRIBUTES:
        isAdmin : Boolean
    
    CONSTRUCTOR Admin(username, password, isAdmin):
        SUPER(username, password)
        this.isAdmin = isAdmin
    
    METHOD login():   // Overriding
        IF validateCredentials(username, password) AND isAdmin == TRUE:
            PRINT "Admin " + username + " logged in. Access exam management tools."
        ELSE:
            PRINT "Login failed. Invalid admin credentials."
    
    METHOD validateCredentials(username, password):
        // Simulated check (real-world: DB check)
        RETURN password == "admin123"
END CLASS


// ---------------- MAIN PROGRAM ----------------
FUNCTION Main():
    // Create objects
    user1 = NEW Student("john123", "student123", TRUE)
    user2 = NEW Admin("profX", "admin123", TRUE)
    user3 = NEW Student("mark99", "wrongpass", TRUE)
    
    // Polymorphic calls
    user1.login()   // Student login success
    user2.login()   // Admin login success
    user3.login()   // Student login fails
END FUNCTION
