ALGORITHM OnlineStoreCart
START
    CLASS Item
        ATTRIBUTES: name, price
        CONSTRUCTOR(name, price)
    END CLASS

    CLASS ShoppingCart
        ATTRIBUTE: cartItems LIST OF Item

        METHOD addItem(item)
            ADD item TO cartItems
        END METHOD

        GENERIC METHOD calculateTotalPrice<T extends Item>()
            total ← 0
            FOR EACH item IN cartItems
                total ← total + item.price
            END FOR
            RETURN total
        END METHOD
    END CLASS
END
